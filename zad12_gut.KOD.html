    <label>Dzień D1:</label>
    <input type="number" id="d1">
    <br><br>

    <label>Godzina G1:</label>
    <input type="number" id="g1">
    <br><br>

    <label>Dzień D2:</label>
    <input type="number" id="d2">
    <br><br>

    <label>Godzina G2:</label>
    <input type="number" id="g2">
    <br><br>

    <button onclick="oblicz()">Oblicz</button>

    <div id="wynik"></div>

    <script>
        function oblicz() {
            let d1 = Number(document.getElementById("d1").value);
            let g1 = Number(document.getElementById("g1").value);
            let d2 = Number(document.getElementById("d2").value);
            let g2 = Number(document.getElementById("g2").value);

            let czas1 = d1 * 24 + g1;
            let czas2 = d2 * 24 + g2;

            let roznicaGodzin = Math.abs(czas1 - czas2);
            let sekundy = roznicaGodzin * 60 * 60;

            document.getElementById("wynik").innerHTML =
                "Liczba sekund: " + sekundy;
        }
    </script>
